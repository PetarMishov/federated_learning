// Logical schema. Executable definitions are in db/schema.sql.
// Lifecycle constraints/functions are in db/deployment_rules.sql.
// GitHub: GitHub App user authorization with read-only repository contents.
// GitLab: OAuth authorization-code flow with repository read access.

Table users {
  id integer [pk]
  username varchar [unique, not null]
  password_hash varchar [not null]
  Note: 'One hospital machine account represents one execution machine.'
}

Table revoked_tokens {
  jti text [pk]
  expires_at timestamptz [not null]
  Note: 'Revoked JWT identifiers persist across API restarts. Expired records may be removed.'
}

Table notifications {
  id integer [pk]
  user_id integer [not null]
  title text [not null]
  message text [not null]
  created_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  read_at timestamptz
  indexes {
    (user_id, created_at)
  }
  Note: 'Each notification belongs to one user. A null read_at means unread. No links to other tables.'
}

Table organizations {
  id integer [pk]
  name varchar [not null]
  owner_user_id integer [not null]
  Note: 'Owner is the predefined all-permissions role, represented by owner_user_id. Only the current owner can transfer ownership. Owner must be a member: enforce transactionally, including organization creation. Ownership is separate from editable custom roles.'
}

Table user_organization {
  user_id integer [not null]
  org_id integer [not null]
  role_id integer
  indexes {
    (user_id, org_id) [pk]
  }
  Note: 'Custom role is optional for an owner or a member without an assigned role.'
}

Table roles {
  id integer [pk]
  org_id integer [not null]
  name varchar [not null]
  indexes {
    (org_id, name) [unique]
    (org_id, id) [unique]
  }
  Note: 'Organization-specific custom roles. Role editors may grant only permissions they possess; enforce in the application.'
}

Table role_permission {
  role_id integer [not null]
  perm_id integer [not null]
  scope permission_scope [not null, default: 'organization']
  indexes {
    (role_id, perm_id) [pk]
  }
}

Table permissions {
  id integer [pk]
  name varchar [unique, not null]
  scope permission_scope [not null]
  indexes {
    (id, scope) [unique]
  }
  Note: 'Include edit_roles, edit_project, start_deployment, and participate_in_deployment. Participation permits and requires inspecting published contents before acceptance.'
}

Enum permission_scope {
  organization
  project
}

Table role_project_permission {
  org_id integer [not null]
  role_id integer [not null]
  project_id integer [not null]
  perm_id integer [not null]
  scope permission_scope [not null, default: 'project']
  indexes {
    (role_id, project_id, perm_id) [pk]
  }
  Note: 'Role and project must belong to the same organization. Only project-scoped permissions belong here; role_permission holds edit_roles only.'
}

Enum provider_kind {
  github
  gitlab
}

Table provider_connections {
  id integer [pk]
  user_id integer [not null]
  provider provider_kind [not null]
  external_account_id varchar [not null]
  external_username varchar
  access_token_encrypted text [not null]
  refresh_token_encrypted text
  token_expires_at timestamp
  granted_permissions text
  created_at timestamp [not null]
  updated_at timestamp [not null]
  invalidated_at timestamp
  indexes {
    (user_id, provider) [unique]
  }
  Note: 'Personal connections. External account ID is deliberately not unique. Protect credentials using keys outside the database; never publish them with code. Local unlink affects only this connection. Provider revocation affects credentials actually revoked, not all connections sharing an account ID. Check GitHub installation access when fetching repositories.'
}

Table projects {
  id integer [pk]
  org_id integer [not null]
  created_by_user_id integer [not null]
  name varchar [not null]
  dockerfile_path varchar
  compose_file_path varchar
  build_context_path varchar [not null, default: '.']
  env_file_path varchar
  command_arguments json
  input_mount_destination varchar [not null]
  output_mount_destination varchar [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
  indexes {
    (org_id, id) [unique]
  }
  Note: 'Editable defaults copied into new runs. Paths refer to snapshot contents. Validate Docker/Compose selection and mount destinations. Changing defaults does not silently modify existing runs.'
}

Enum snapshot_source {
  github
  gitlab
  local
  platform
}

Table snapshots {
  id integer [pk]
  project_id integer [not null]
  created_by_user_id integer [not null]
  source snapshot_source [not null]
  source_repository_id varchar
  source_repository_url varchar
  source_branch varchar
  source_commit_sha varchar
  storage_key varchar [unique, not null]
  content_sha256 varchar [not null]
  created_at timestamp [not null]
  indexes {
    (project_id, id) [unique]
  }
  Note: 'Immutable exact project files at one revision, without .git/history. Files live in local artifact storage and hashes identify the archives. Source fields are provenance, not access grants. Snapshot survives unlinking. Publish contents to eligible organization members when a run becomes pending; provider repository browsing remains personal.'
}

Enum run_status {
  pending
  started
  finished
  cancelled
  failed
  skipped
}

Table deployment_runs {
  id integer [pk]
  org_id integer [not null]
  project_id integer [not null]
  snapshot_id integer [not null]
  name text [not null, default: 'Deployment']
  change_summary text [not null]
  supersedes_run_id integer [unique]
  terminal_reason text
  cancelled_by_user_id integer
  created_by_user_id integer [not null]
  started_by_user_id integer
  status run_status [not null, default: 'pending']
  dockerfile_path varchar
  compose_file_path varchar
  build_context_path varchar [not null]
  env_file_path varchar
  command_arguments json
  input_mount_destination varchar [not null]
  output_mount_destination varchar [not null]
  coordinator_session_id varchar [unique]
  coordinator_endpoint varchar
  created_at timestamp [not null]
  started_at timestamp
  ended_at timestamp
  indexes {
    (project_id, id) [unique]
    (id, snapshot_id) [unique]
  }
  Note: 'Explicit pending updates create a replacement and skip the predecessor. Snapshot/settings remain immutable. Starter must join with at least one other participant. One remaining participant continues; zero cancels. Pending/started can be cancelled. Finished/failed/cancelled/skipped are terminal.'
}

Enum participation_status {
  pending
  joined
  removed
}

Enum removal_reason {
  voluntary
  permission_lost
  unauthorized_behavior
}

Enum execution_status {
  not_started
  starting
  running
  completed
  failed
  stopped
}

Table run_participants {
  run_id integer [not null]
  user_id integer [not null]
  participation_status participation_status [not null, default: 'pending']
  accepted_snapshot_id integer
  accepted_at timestamp
  local_input_path varchar
  local_output_path varchar
  execution_status execution_status [not null, default: 'not_started']
  joined_at timestamp
  removed_at timestamp
  removal_reason removal_reason
  stop_requested_at timestamp
  stopped_at timestamp
  indexes {
    (run_id, user_id) [pk]
  }
  Note: 'Pending users are eligible but have not accepted. Joined users accept this deployment snapshot. Joining/rejoining requires pending status. Removed covers voluntary departure, permission loss, or future unauthorized-behavior detection. Historical rows and stop acknowledgments survive termination. Local path edits do not invalidate shared acceptance.'
}
