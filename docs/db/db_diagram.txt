// Logical schema. Executable definitions are in db/schema.sql.
// Lifecycle constraints/functions are in db/deployment_rules.sql.
// Types, defaults, indexes, and foreign keys mirror db/schema.sql.
// CHECK constraints and deferred constraints are described in table notes.

Table users {
  id integer [pk, increment]
  username text [unique, not null]
  password_hash text [not null]
  Note: 'One hospital machine account represents one execution machine.'
}

Table revoked_tokens {
  jti text [pk]
  expires_at timestamptz [not null]
  Note: 'Revoked JWT identifiers persist across API restarts. Expired records may be removed.'
}

Table notifications {
  id integer [pk, increment]
  user_id integer [not null]
  title text [not null]
  message text [not null]
  created_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  read_at timestamptz
  indexes {
    (user_id, created_at) [name: 'notifications_user_created_idx'] // created_at DESC in PostgreSQL
  }
  Note: 'Each notification belongs to one user. A null read_at means unread. Deployment updates are described in message; there is no deployment foreign key.'
}

Table organizations {
  id integer [pk, increment]
  name text [not null]
  owner_user_id integer [not null]
  indexes {
    owner_user_id [name: 'organizations_owner_idx']
  }
  Note: 'Owner membership is enforced by a DEFERRABLE INITIALLY DEFERRED composite foreign key. Owner is the predefined all-permissions role, represented by owner_user_id. Only the current owner can transfer ownership. Owner must be a member: enforce transactionally, including organization creation. Ownership is separate from editable custom roles.'
}

Table user_organization {
  user_id integer [not null]
  org_id integer [not null]
  role_id integer
  indexes {
    (user_id, org_id) [pk]
    (org_id, role_id) [name: 'memberships_org_role_idx']
  }
  Note: 'Custom role is optional for an owner or a member without an assigned role.'
}

Table roles {
  id integer [pk, increment]
  org_id integer [not null]
  name text [not null]
  indexes {
    (org_id, name) [unique]
    (org_id, id) [unique]
  }
  Note: 'Organization-specific custom roles. Role editors may grant only permissions they possess; enforce in the application.'
}

Table role_permission {
  role_id integer [not null]
  perm_id integer [not null]
  scope permission_scope [not null, default: 'organization']
  indexes {
    (role_id, perm_id) [pk]
    perm_id [name: 'role_permission_permission_idx']
  }
  Note: 'CHECK requires organization scope; the composite permission foreign key enforces the catalog scope.'
}

Table permissions {
  id integer [pk, increment]
  name text [unique, not null]
  scope permission_scope [not null]
  indexes {
    (id, scope) [unique]
  }
  Note: 'Fixed catalog: edit_roles has organization scope; edit_project, start_deployment, and participate_in_deployment have project scope. A CHECK rejects other names or scopes. Participation permits and requires inspecting published contents before acceptance.'
}

Enum permission_scope {
  organization
  project
}

Table role_project_permission {
  org_id integer [not null]
  role_id integer [not null]
  project_id integer [not null]
  perm_id integer [not null]
  scope permission_scope [not null, default: 'project']
  indexes {
    (role_id, project_id, perm_id) [pk]
    (project_id, role_id) [name: 'project_grants_project_idx']
  }
  Note: 'Role and project must belong to the same organization. A CHECK requires project scope; role_permission holds edit_roles only.'
}

Enum provider_kind {
  github
  gitlab
}

Table provider_connections {
  id integer [pk, increment]
  user_id integer [not null]
  provider provider_kind [not null]
  external_account_id text [not null]
  external_username text
  access_token_encrypted text [not null]
  refresh_token_encrypted text
  token_expires_at timestamptz
  granted_permissions text
  created_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  updated_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  invalidated_at timestamptz
  indexes {
    (user_id, provider) [unique]
  }
  Note: 'Personal connections. External account ID is deliberately not unique. Protect credentials using keys outside the database; never publish them with code. Local unlink affects only this connection. Provider revocation affects credentials actually revoked, not all connections sharing an account ID. Check GitHub installation access when fetching repositories.'
}

Table projects {
  id integer [pk, increment]
  org_id integer [not null]
  created_by_user_id integer [not null]
  name text [not null]
  dockerfile_path text
  compose_file_path text
  build_context_path text [not null, default: '.']
  env_file_path text
  command_arguments jsonb
  input_mount_destination text [not null]
  output_mount_destination text [not null]
  created_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  updated_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  indexes {
    (org_id, id) [unique]
    created_by_user_id [name: 'projects_creator_idx']
  }
  Note: 'Editable defaults copied into new runs. Paths refer to snapshot contents. Validate Docker/Compose selection and mount destinations. Changing defaults does not silently modify existing runs.'
}

Enum snapshot_source {
  github
  gitlab
  local
  platform
}

Table snapshots {
  id integer [pk, increment]
  project_id integer [not null]
  created_by_user_id integer [not null]
  source snapshot_source [not null]
  source_repository_id text
  source_repository_url text
  source_branch text
  source_commit_sha text
  storage_key text [unique, not null]
  content_sha256 text [not null]
  created_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  indexes {
    (project_id, id) [unique]
    created_by_user_id [name: 'snapshots_creator_idx']
  }
  Note: 'CHECK constraints require a 64-character hexadecimal content hash and a relative storage key without parent traversal. Immutable exact project files at one revision, without .git/history. Files live in local artifact storage and hashes identify the archives. Source fields are provenance, not access grants. Snapshot survives unlinking. Publish contents to eligible organization members when a run becomes pending; provider repository browsing remains personal.'
}

Enum run_status {
  pending
  started
  finished
  cancelled
  failed
  skipped
}

Table deployment_runs {
  id integer [pk, increment]
  org_id integer [not null]
  project_id integer [not null]
  snapshot_id integer [not null]
  name text [not null, default: 'Deployment']
  change_summary text [not null, default: '']
  supersedes_run_id integer [unique]
  terminal_reason text
  cancelled_by_user_id integer
  created_by_user_id integer [not null]
  started_by_user_id integer
  status run_status [not null, default: 'pending']
  dockerfile_path text
  compose_file_path text
  build_context_path text [not null]
  env_file_path text
  command_arguments jsonb
  input_mount_destination text [not null]
  output_mount_destination text [not null]
  coordinator_session_id text [unique]
  coordinator_endpoint text
  created_at timestamptz [not null, default: `CURRENT_TIMESTAMP`]
  started_at timestamptz
  ended_at timestamptz
  indexes {
    (project_id, id) [unique]
    (id, snapshot_id) [unique]
    (project_id, snapshot_id) [name: 'runs_project_snapshot_idx']
    (org_id, status) [name: 'runs_org_status_idx']
    created_by_user_id [name: 'runs_creator_idx']
    started_by_user_id [name: 'runs_starter_idx']
  }
  Note: 'CHECK constraints pair starter/start time, require start metadata for started/finished/failed, forbid start metadata for pending/skipped, require end time exactly for terminal states, order creation/start/end times, require a nonblank failure reason, restrict cancellation actors to cancelled runs, and forbid self-replacement. Explicit pending updates create a replacement and skip the predecessor. Snapshot/settings remain immutable. Starter must join with at least one other participant. One remaining participant continues; zero cancels. Pending/started can be cancelled. Finished/failed/cancelled/skipped are terminal.'
}

Enum participation_status {
  pending
  joined
  removed
}

Enum removal_reason {
  voluntary
  permission_lost
  unauthorized_behavior
}

Enum execution_status {
  not_started
  starting
  running
  completed
  failed
  stopped
}

Table run_participants {
  run_id integer [not null]
  user_id integer [not null]
  participation_status participation_status [not null, default: 'pending']
  accepted_snapshot_id integer
  accepted_at timestamptz
  local_input_path text
  local_output_path text
  execution_status execution_status [not null, default: 'not_started']
  joined_at timestamptz
  removed_at timestamptz
  removal_reason removal_reason
  stop_requested_at timestamptz
  stopped_at timestamptz
  indexes {
    (run_id, user_id) [pk]
    user_id [name: 'participants_user_idx']
  }
  Note: 'CHECK constraints pair snapshot acceptance/time and removal reason/time. Joined requires acceptance, join time, and nonblank input/output paths. Pending requires no acceptance/join time and not_started execution. Pending users are eligible but have not accepted. Joined users accept this deployment snapshot. Joining/rejoining requires the deployment to be pending and the user to be eligible. Removed covers voluntary departure, permission loss, or future unauthorized-behavior detection. Historical rows and stop acknowledgments survive termination. Local path edits do not invalidate shared acceptance.'
}
